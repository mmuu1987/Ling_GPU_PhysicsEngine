using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text;
using UnityEngine;
using UnityEngine.UI;
using Debug = UnityEngine.Debug;
using Object = UnityEngine.Object;

namespace MassEngine.Game
{
    /// <summary>
    /// Temporary P12-only Player harness: Windows x64 scale test on the 1024 m candidate map forest-green22-1024 (100k/200k, D3D11 vs D3D12), Burst nav default. Strict layout: density 0.7, aspect 2.2, gap 50, z 0.
    /// Inert unless launched with --p3s-scale-validation. Not shipping code; removed after the run.
    /// </summary>
    public sealed class P3ScaleFrameHarness : MonoBehaviour
    {
        [Serializable] sealed class Nav
        {
            public bool found; public string preparationState, preparationBlockReason;
            public int completedFields, burstSolves, managedSolves, managedNativeSolves, probes;
            public double totalSolveMs, maxSolveMs, lastSolveMs, maxProbeMs;
        }
        [Serializable] sealed class Result
        {
            public bool passed, managedArgument, deploymentValid, startConfirmClicked, sampledWhileRunning;
            public string error, unityVersion, graphicsApi, graphicsDeviceName, graphicsDeviceVersion, platform, scene;
            public string phaseAtSampleStart, phaseAtSampleEnd, deploymentNote, navBackend, terrainId;
            public float worldSizeX, worldSizeZ, navCellSize, flowCellSize, surfaceStep; public int navResolutionX, navResolutionZ, flowResolution;
            public int requestedUnits, perSide, draftEntries, configuredUnits, agentCount, aliveAtSampleStart, aliveAtSampleEnd;
            public int team0Initial, team1Initial, layoutAttempts;
            public float densityUsed, aspectUsed, gapUsed, zOffsetUsed;
            public int screenWidth, screenHeight, vSyncCount, targetFrameRate, frames, over33, over50, graphicsMemorySizeMB;
            public double warmupSeconds, sampleSeconds, actualSampleSeconds, medianMs, p95Ms, p99Ms, maxMs, meanMs, avgFps;
            public double loadSeconds, applyToRunningSeconds;
            public long gfxDriverBytesAtSampleStart, gfxDriverBytesPeak, gfxDriverBytesEnd, totalAllocatedBytesEnd, totalReservedBytesEnd;
            public Nav navAtSampleStart = new Nav(), navAtSampleEnd = new Nav();
            public List<string> stages = new List<string>();
            public List<string> validationErrors = new List<string>();
        }

        const string EntryId = "forest-green22-1024";
        const double WarmupSeconds = 20, SampleSeconds = 60, HardLimitSeconds = 600;
        static readonly float[] Densities = { .7f }; const float Aspect = 2.2f;
        static readonly float[] Gaps = { 50f };
        static readonly float[] ZOffsets = { 0f };
        static string output; static int requested; static bool booted;
        readonly Result result = new Result();
        WarSandboxSceneSession session; WarSandboxBattleController controller; MassEngineManager manager;
        readonly List<double> frameMs = new List<double>(16384);
        bool sampling, trackMemory; long lastTick; double hardDeadline, nextBeat, nextMem;

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
        static void Bootstrap()
        {
            if (booted) return;
            var args = Environment.GetCommandLineArgs();
            if (!args.Contains("--p3s-scale-validation")) return;
            string arg = args.FirstOrDefault(x => x.StartsWith("--p3s-output=", StringComparison.Ordinal));
            string units = args.FirstOrDefault(x => x.StartsWith("--p3s-units=", StringComparison.Ordinal));
            if (arg == null || units == null) return;
            output = arg.Substring("--p3s-output=".Length).Trim('"');
            if (string.IsNullOrWhiteSpace(output) || !int.TryParse(units.Substring("--p3s-units=".Length), out requested)) return;
            booted = true;
            var host = new GameObject("P3 scale D3D11/D3D12 validation (temporary)");
            Object.DontDestroyOnLoad(host);
            host.AddComponent<P3ScaleFrameHarness>();
        }

        void Start()
        {
            Directory.CreateDirectory(output);
            var args = Environment.GetCommandLineArgs();
            result.managedArgument = args.Contains("--war-sandbox-nav-managed");
            result.requestedUnits = requested; result.perSide = requested / 2;
            result.unityVersion = Application.unityVersion;
            result.graphicsApi = SystemInfo.graphicsDeviceType.ToString();
            result.graphicsDeviceName = SystemInfo.graphicsDeviceName;
            result.graphicsDeviceVersion = SystemInfo.graphicsDeviceVersion;
            result.graphicsMemorySizeMB = SystemInfo.graphicsMemorySize;
            result.platform = Application.platform.ToString();
            result.warmupSeconds = WarmupSeconds; result.sampleSeconds = SampleSeconds;
            hardDeadline = Time.realtimeSinceStartupAsDouble + HardLimitSeconds;
            QualitySettings.vSyncCount = 0; Application.targetFrameRate = -1;
            Stage("harness-started;api=" + result.graphicsApi + ";units=" + requested);
            StartCoroutine(Guarded());
        }

        void Update()
        {
            long now = Stopwatch.GetTimestamp();
            if (sampling && lastTick != 0) frameMs.Add((now - lastTick) * 1000.0 / Stopwatch.Frequency);
            lastTick = now;
            double t = Time.realtimeSinceStartupAsDouble;
            if (trackMemory && t >= nextMem)
            {
                nextMem = t + 0.5;
                long g = UnityEngine.Profiling.Profiler.GetAllocatedMemoryForGraphicsDriver();
                if (g > result.gfxDriverBytesPeak) result.gfxDriverBytesPeak = g;
            }
            if (!sampling && t >= nextBeat) { nextBeat = t + 5; Beat(); }
            if (!result.passed && string.IsNullOrEmpty(result.error) && t > hardDeadline)
                Finish(false, "Exceeded " + HardLimitSeconds + " s.");
        }

        void Beat()
        {
            try { File.WriteAllText(Path.Combine(output, "progress.txt"), DateTime.Now.ToString("o") + " " + (result.stages.Count > 0 ? result.stages[result.stages.Count - 1] : "")); } catch { }
        }

        void Stage(string s)
        {
            result.stages.Add(s + ";t_s=" + Time.realtimeSinceStartupAsDouble.ToString("F3", CultureInfo.InvariantCulture));
            Debug.Log("P3S_PLAYER_STAGE " + s);
            Beat();
        }

        static Button FindButton(string name)
        {
            return Object.FindObjectsByType<Button>(FindObjectsSortMode.None).FirstOrDefault(x => x != null && x.gameObject.activeInHierarchy && x.name == name && x.interactable);
        }

        IEnumerator Guarded()
        {
            var e = Run();
            while (true)
            {
                object next;
                try { if (!e.MoveNext()) break; next = e.Current; }
                catch (Exception ex) { sampling = false; Finish(false, "Exception: " + ex); yield break; }
                yield return next;
            }
        }

        IEnumerator Run()
        {
            if (Application.platform != RuntimePlatform.WindowsPlayer) { Finish(false, "Expected WindowsPlayer, got " + Application.platform); yield break; }
            if (requested != 100000 && requested != 200000) { Finish(false, "Unapproved population " + requested); yield break; }
            double until = Time.realtimeSinceStartupAsDouble + 120;
            while ((session = WarSandboxSceneSession.Instance) == null && Time.realtimeSinceStartupAsDouble < until) yield return null;
            if (session == null) { Finish(false, "MainMenu session missing."); yield break; }
            while (session.IsLoading && Time.realtimeSinceStartupAsDouble < until) yield return null;
            if (session.State != WarSandboxEntryState.Menu) { Finish(false, "Expected Menu, got " + session.State + ": " + session.Error); yield break; }
            Stage("menu-ready");
            double loadStart = Time.realtimeSinceStartupAsDouble;
            if (!session.TryEnterBattlefield(EntryId, false, out string error)) { Finish(false, "TryEnterBattlefield: " + error); yield break; }
            until = Time.realtimeSinceStartupAsDouble + 180;
            while (session.IsLoading && Time.realtimeSinceStartupAsDouble < until) yield return null;
            if (session.IsLoading || session.State != WarSandboxEntryState.Battle) { Finish(false, "Entry failed: " + session.State + " / " + session.Error); yield break; }
            result.loadSeconds = Time.realtimeSinceStartupAsDouble - loadStart;
            controller = session.Controller; manager = controller != null ? controller.manager : null;
            if (manager == null) { Finish(false, "Manager missing."); yield break; }
            result.scene = session.CurrentEntryId;
            try
            {
                var sim = manager.systemConfig.simulationConfig; var flow = manager.systemConfig.runtimeFlowConfig;
                result.worldSizeX = sim.simulationWorldSize.x; result.worldSizeZ = sim.simulationWorldSize.y;
                result.flowResolution = flow.flowFieldResolution; result.flowCellSize = flow.flowFieldCellSize;
                if (manager.TryGetTerrainContext(out var surf, out var navg, out _) && surf != null && navg != null)
                { result.terrainId = surf.Id; result.surfaceStep = surf.Step.x; result.navCellSize = navg.CellSize; result.navResolutionX = navg.ResolutionX; result.navResolutionZ = navg.ResolutionZ; }
            }
            catch (Exception ex) { result.terrainId = "read-error:" + ex.Message; }
            Stage("entered;defaultAgents=" + (manager.Buffers != null ? manager.Buffers.AgentCount : -1) + ";world=" + result.worldSizeX + ";nav=" + result.navResolutionX + "@" + result.navCellSize + ";terrain=" + result.terrainId);
            if (result.worldSizeX != 1024f || result.terrainId != "forest-hills-scale15-1024") { Finish(false, "Not the 1024 m candidate map: world=" + result.worldSizeX + " terrain=" + result.terrainId); yield break; }
            for (int i = 0; i < 30; i++) yield return null;

            var d = controller.GetComponent<WarSandboxRuntimeDeployment>();
            if (d == null) { Finish(false, "WarSandboxRuntimeDeployment missing."); yield break; }
            until = Time.realtimeSinceStartupAsDouble + 5;
            while (!d.IsEditing && Time.realtimeSinceStartupAsDouble < until) yield return null;
            if (!d.IsEditing)
            {
                if (!d.TryBeginEdit(false, out error)) { Finish(false, "TryBeginEdit: " + error + " / phase=" + controller.Phase); yield break; }
                Stage("begin-edit-called;phase=" + controller.Phase);
                for (int i = 0; i < 30; i++) yield return null;
            }
            if (!d.IsEditing || d.Draft == null) { Finish(false, "Deployment editor not active after TryBeginEdit."); yield break; }
            result.draftEntries = d.Draft.Count;
            if (d.Draft.Count != 2) { Finish(false, "Expected 2 draft entries, got " + d.Draft.Count); yield break; }
            var base0 = d.Draft[0]; var base1 = d.Draft[1];
            int per = requested / 2; bool fit = false; int attempts = 0;
            foreach (float density in Densities)
            {
                foreach (float gap in Gaps)
                {
                    foreach (float z in ZOffsets)
                    {
                        var a = base0; var b = base1;
                        a.count = per; b.count = per; a.manualSize = Vector3.zero; b.manualSize = Vector3.zero;
                        a.density = density; b.density = density; a.aspect = Aspect; b.aspect = Aspect;
                        int t0 = a.teamId == 0 ? -1 : 1, t1 = b.teamId == 0 ? -1 : 1;
                        if (t0 == t1) { t0 = -1; t1 = 1; }
                        a.center = new Vector3(t0 * (gap * .5f + a.Size.x * .5f), base0.center.y, z);
                        b.center = new Vector3(t1 * (gap * .5f + b.Size.x * .5f), base1.center.y, z);
                        d.Draft.Set(0, a); d.Draft.Set(1, b); attempts++;
                        if (d.TryValidate(out error))
                        {
                            fit = true; result.densityUsed = density; result.aspectUsed = a.aspect; result.gapUsed = gap; result.zOffsetUsed = z;
                            break;
                        }
                        if (result.validationErrors.Count < 40) result.validationErrors.Add("d=" + density + ";gap=" + gap + ";z=" + z + ";size=" + a.Size + ": " + error);
                        if (attempts % 8 == 0) yield return null;
                    }
                    if (fit) break;
                }
                if (fit) break;
            }
            result.layoutAttempts = attempts;
            result.deploymentValid = fit;
            if (!fit) { Finish(false, "No valid deployment layout for " + requested + " units on " + EntryId + " (strict density .7 / aspect 2.2 / gap 50 / z 0)."); yield break; }
            result.deploymentNote = "density=" + result.densityUsed + ";aspect=" + result.aspectUsed + ";gap=" + result.gapUsed + ";z=" + result.zOffsetUsed + ";size=" + d.Draft[0].Size;
            Stage("layout-valid;" + result.deploymentNote);
            yield return null; yield return null;

            Button start = null;
            until = Time.realtimeSinceStartupAsDouble + 5;
            while ((start = FindButton("deployment-start")) == null && Time.realtimeSinceStartupAsDouble < until) yield return null;
            double applyStart = Time.realtimeSinceStartupAsDouble;
            if (start != null) start.onClick.Invoke();
            else
            {
                Stage("deployment-start-button-missing;fallback=TryApply");
                if (!d.TryApply(out error)) { Finish(false, "TryApply: " + error); yield break; }
            }
            until = Time.realtimeSinceStartupAsDouble + 10;
            while (controller.Phase != WarSandboxBattlePhase.Running && Time.realtimeSinceStartupAsDouble < until)
            {
                var confirm = FindButton("tight-dialog-confirm");
                if (confirm != null) { confirm.onClick.Invoke(); result.startConfirmClicked = true; Stage("tight-dialog-confirm"); }
                yield return null;
            }
            if (controller.Phase != WarSandboxBattlePhase.Running)
            {
                controller.StartOrResumeBattle();
                until = Time.realtimeSinceStartupAsDouble + 30;
                while (controller.Phase != WarSandboxBattlePhase.Running && Time.realtimeSinceStartupAsDouble < until) yield return null;
            }
            if (controller.Phase != WarSandboxBattlePhase.Running) { Finish(false, "Battle not Running: " + controller.Phase + " / " + d.Error + " / " + controller.CommandError); yield break; }
            result.applyToRunningSeconds = Time.realtimeSinceStartupAsDouble - applyStart;
            result.agentCount = manager.Buffers != null ? manager.Buffers.AgentCount : -1;
            try { result.configuredUnits = manager.scenarioConfig.unitTypes.Sum(u => u.spawnConfig.unitCount); } catch { result.configuredUnits = -1; }
            try { result.team0Initial = controller.GetArmy(0).initialUnitCount; result.team1Initial = controller.GetArmy(1).initialUnitCount; } catch { }
            double battleStart = Time.realtimeSinceStartupAsDouble;
            trackMemory = true;
            Stage("battle-running;agents=" + result.agentCount);
            if (result.agentCount != requested) { Finish(false, "Expected " + requested + " agents, got " + result.agentCount); yield break; }
            while (Time.realtimeSinceStartupAsDouble - battleStart < WarmupSeconds) yield return null;
            result.phaseAtSampleStart = controller.Phase.ToString();
            result.navAtSampleStart = ReadNav();
            result.aliveAtSampleStart = Alive();
            result.gfxDriverBytesAtSampleStart = UnityEngine.Profiling.Profiler.GetAllocatedMemoryForGraphicsDriver();
            Stage("sample-start;alive=" + result.aliveAtSampleStart + ";nav=" + result.navAtSampleStart.preparationState);
            yield return null;
            lastTick = 0; sampling = true;
            double sampleStart = Time.realtimeSinceStartupAsDouble;
            bool allRunning = true;
            while (Time.realtimeSinceStartupAsDouble - sampleStart < SampleSeconds)
            {
                if (controller.Phase != WarSandboxBattlePhase.Running) allRunning = false;
                yield return null;
            }
            sampling = false;
            result.actualSampleSeconds = Time.realtimeSinceStartupAsDouble - sampleStart;
            trackMemory = false;
            result.sampledWhileRunning = allRunning;
            result.phaseAtSampleEnd = controller.Phase.ToString();
            result.navAtSampleEnd = ReadNav();
            result.gfxDriverBytesEnd = UnityEngine.Profiling.Profiler.GetAllocatedMemoryForGraphicsDriver();
            if (result.gfxDriverBytesEnd > result.gfxDriverBytesPeak) result.gfxDriverBytesPeak = result.gfxDriverBytesEnd;
            result.totalAllocatedBytesEnd = UnityEngine.Profiling.Profiler.GetTotalAllocatedMemoryLong();
            result.totalReservedBytesEnd = UnityEngine.Profiling.Profiler.GetTotalReservedMemoryLong();
            result.aliveAtSampleEnd = Alive();
            var ns = result.navAtSampleStart; var ne = result.navAtSampleEnd;
            result.navBackend = !ne.found ? "not-found" : (ne.burstSolves > ns.burstSolves && ne.managedSolves == ns.managedSolves ? "Burst" : (ne.managedSolves > ns.managedSolves && ne.burstSolves == ns.burstSolves ? "Managed" : (ne.burstSolves == ns.burstSolves && ne.managedSolves == ns.managedSolves ? "no-solves-in-window(burstTotal=" + ne.burstSolves + ",managedTotal=" + ne.managedSolves + ")" : "Mixed")));
            Stage("sample-end;frames=" + frameMs.Count);
            Summarize();
            try { ScreenCapture.CaptureScreenshot(Path.Combine(output, "end.png")); } catch { }
            for (int i = 0; i < 5; i++) yield return null;
            string invalid = null;
            if (result.phaseAtSampleStart != "Running" || !allRunning) invalid = "Battle not Running for the whole window.";
            Finish(invalid == null, invalid);
        }

        int Alive()
        {
            try
            {
                int n = manager.Buffers.AgentCount;
                var hp = new int[n];
                manager.Buffers.combatBuffers.hpReadBuffer.GetData(hp, 0, 0, n);
                int alive = 0; for (int i = 0; i < n; i++) if (hp[i] > 0) alive++;
                return alive;
            }
            catch (Exception ex) { Debug.LogWarning("P3S alive read failed: " + ex.Message); return -1; }
        }

        Nav ReadNav()
        {
            var n = new Nav();
            try
            {
                var f = typeof(MassEngineManager).GetField("terrainRuntime", BindingFlags.Instance | BindingFlags.NonPublic);
                object rt = f != null ? f.GetValue(manager) : null;
                if (rt == null) return n;
                n.found = true;
                Type t = rt.GetType();
                object P(string name) { var p = t.GetProperty(name, BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic); return p != null ? p.GetValue(rt) : null; }
                n.preparationState = P("PreparationState") as string ?? "n/a";
                n.preparationBlockReason = P("PreparationBlockReason") as string;
                n.completedFields = Convert.ToInt32(P("CompletedFields") ?? -1);
                n.burstSolves = Convert.ToInt32(P("BurstSolveCount") ?? -1);
                n.managedSolves = Convert.ToInt32(P("ManagedSolveCount") ?? -1);
                n.managedNativeSolves = Convert.ToInt32(P("ManagedNativeSolveCount") ?? -1);
                n.probes = Convert.ToInt32(P("PreparationProbeCount") ?? -1);
                n.totalSolveMs = Convert.ToDouble(P("TotalSolveMilliseconds") ?? -1.0);
                n.maxSolveMs = Convert.ToDouble(P("MaxSolveMilliseconds") ?? -1f);
                n.lastSolveMs = Convert.ToDouble(P("LastSolveMilliseconds") ?? -1f);
                n.maxProbeMs = Convert.ToDouble(P("MaxPreparationProbeMilliseconds") ?? -1.0);
            }
            catch (Exception ex) { n.preparationBlockReason = "read-error:" + ex.GetType().Name + ":" + ex.Message; }
            return n;
        }

        void Summarize()
        {
            result.frames = frameMs.Count;
            result.screenWidth = Screen.width; result.screenHeight = Screen.height;
            result.vSyncCount = QualitySettings.vSyncCount; result.targetFrameRate = Application.targetFrameRate;
            if (frameMs.Count == 0) return;
            var s = frameMs.OrderBy(x => x).ToArray();
            double Q(double q) { double pos = (s.Length - 1) * q; int lo = (int)Math.Floor(pos), hi = (int)Math.Ceiling(pos); return s[lo] + (s[hi] - s[lo]) * (pos - lo); }
            result.medianMs = Q(.5); result.p95Ms = Q(.95); result.p99Ms = Q(.99); result.maxMs = s[s.Length - 1]; result.meanMs = s.Average();
            result.over33 = s.Count(x => x > 33.33); result.over50 = s.Count(x => x > 50);
            result.avgFps = result.actualSampleSeconds > 0 ? frameMs.Count / result.actualSampleSeconds : 0;
            var sb = new StringBuilder("index,ms\n");
            for (int i = 0; i < frameMs.Count; i++) sb.Append(i).Append(',').Append(frameMs[i].ToString("R", CultureInfo.InvariantCulture)).Append('\n');
            File.WriteAllText(Path.Combine(output, "frames.csv"), sb.ToString());
        }

        void Finish(bool ok, string error)
        {
            if (result.passed || !string.IsNullOrEmpty(result.error)) return;
            result.passed = ok; result.error = ok ? null : (error ?? "unknown");
            try { File.WriteAllText(Path.Combine(output, "result.json"), JsonUtility.ToJson(result, true)); } catch (Exception ex) { Debug.LogError("P3S result write failed: " + ex); }
            Debug.Log("P3S_PLAYER_RESULT passed=" + ok + " error=" + result.error);
            Application.Quit(ok ? 0 : 3);
        }
    }
}
