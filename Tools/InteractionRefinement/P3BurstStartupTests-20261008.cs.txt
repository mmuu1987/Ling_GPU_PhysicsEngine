
namespace MassEngine.Tests {
 public sealed class P3BurstStartupTests {
  const System.Reflection.BindingFlags F=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;
  [Serializable] public sealed class CacheFile {public string path;public long bytes;public string lastWriteUtc;}
  [Serializable] public sealed class CacheView {public string stage;public string utc;public CacheFile[] files;}
  [Serializable] public sealed class Result {public string utcStart,utcEnd;public bool enabled,firstUsedBurst,secondUsedBurst,exactBits,disposed;public double hydrateMs,managedFirstMs,factoryMs,firstRunMs,disposeMs,secondFactoryMs,secondFirstRunMs;public double[] reuseMs;public int created,disposedCount,active,burstCalls,managedCalls;public CacheView[] cacheViews;}
  static string Output(){foreach(var a in Environment.GetCommandLineArgs())if(a.StartsWith("--interaction-p8-output="))return a.Substring("--interaction-p8-output=".Length);throw new Exception("Missing output");}
  static double Ms(long start)=>(System.Diagnostics.Stopwatch.GetTimestamp()-start)*1000.0/System.Diagnostics.Stopwatch.Frequency;
  static CacheView Cache(string stage){var folder=System.IO.Path.GetFullPath(System.IO.Path.Combine(Application.dataPath,"../Library/BurstCache/JIT"));var files=new List<CacheFile>();if(System.IO.Directory.Exists(folder))foreach(var p in System.IO.Directory.GetFiles(folder,"*",System.IO.SearchOption.AllDirectories)){var f=new System.IO.FileInfo(p);files.Add(new CacheFile{path=p.Substring(folder.Length+1),bytes=f.Length,lastWriteUtc=f.LastWriteTimeUtc.ToString("o")});}return new CacheView{stage=stage,utc=DateTime.UtcNow.ToString("o"),files=files.ToArray()};}
  static void Exact(Vector2[] a,Vector2[] b){Assert.AreEqual(a.Length,b.Length);for(int i=0;i<a.Length;i++)if(BitConverter.SingleToInt32Bits(a[i].x)!=BitConverter.SingleToInt32Bits(b[i].x)||BitConverter.SingleToInt32Bits(a[i].y)!=BitConverter.SingleToInt32Bits(b[i].y))Assert.Fail("Float bits "+i);}
  [Test] public void FirstUseAndRecreatedWorkspace(){
   var result=new Result{utcStart=DateTime.UtcNow.ToString("o")};var views=new List<CacheView>();views.Add(Cache("test-entry"));result.enabled=Unity.Burst.BurstCompiler.IsEnabled;Assert.IsTrue(result.enabled);
   string capture=System.IO.Path.GetFullPath(System.IO.Path.Combine(Application.dataPath,"../Logs/InteractionRefinement-20261007/P3/fixed-input-20261008-01-capture"));Vector2[] goals,expected;float radius;
   using(var r=new System.IO.BinaryReader(System.IO.File.OpenRead(System.IO.Path.Combine(capture,"fixed-sample-0-input.bin")))){Assert.AreEqual("P3_FIXED_INPUT_V1",r.ReadString());radius=r.ReadSingle();goals=new Vector2[r.ReadInt32()];for(int i=0;i<goals.Length;i++)goals[i]=new Vector2(r.ReadSingle(),r.ReadSingle());}
   using(var r=new System.IO.BinaryReader(System.IO.File.OpenRead(System.IO.Path.Combine(capture,"fixed-sample-0-expected.bin")))){expected=new Vector2[r.ReadInt32()];for(int i=0;i<expected.Length;i++)expected[i]=new Vector2(r.ReadSingle(),r.ReadSingle());}
   long start=System.Diagnostics.Stopwatch.GetTimestamp();var nav=(TerrainNavigationGrid)typeof(P3UniformOutputTests).GetMethod("Hydrate",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Static).Invoke(null,new object[]{typeof(TerrainNavigationGrid)});result.hydrateMs=Ms(start);
   start=System.Diagnostics.Stopwatch.GetTimestamp();var managed=nav.CreateFlowField(goals,radius);result.managedFirstMs=Ms(start);Exact(expected,managed);
   var factory=typeof(TerrainNavigationGrid).GetMethod("CreateBurstWorkspace",F);Assert.NotNull(factory);object workspace=null;Type type=null;
   try{
    start=System.Diagnostics.Stopwatch.GetTimestamp();workspace=factory.Invoke(nav,null);result.factoryMs=Ms(start);type=workspace.GetType();
    var solve=(Func<IReadOnlyList<Vector2>,float,Vector2[]>)Delegate.CreateDelegate(typeof(Func<IReadOnlyList<Vector2>,float,Vector2[]>),workspace,type.GetMethod("Solve"));
    views.Add(Cache("before-first-run"));start=System.Diagnostics.Stopwatch.GetTimestamp();var actual=solve(goals,radius);result.firstRunMs=Ms(start);result.firstUsedBurst=(bool)type.GetProperty("LastRunUsedBurst").GetValue(workspace);Assert.IsTrue(result.firstUsedBurst);Exact(expected,actual);views.Add(Cache("after-first-run"));
    result.reuseMs=new double[12];for(int i=0;i<12;i++){start=System.Diagnostics.Stopwatch.GetTimestamp();actual=solve(goals,radius);result.reuseMs[i]=Ms(start);Exact(expected,actual);Assert.IsTrue((bool)type.GetProperty("LastRunUsedBurst").GetValue(workspace));}
    start=System.Diagnostics.Stopwatch.GetTimestamp();((IDisposable)workspace).Dispose();result.disposeMs=Ms(start);Assert.IsTrue((bool)type.GetProperty("IsDisposed").GetValue(workspace));workspace=null;
    start=System.Diagnostics.Stopwatch.GetTimestamp();workspace=factory.Invoke(nav,null);result.secondFactoryMs=Ms(start);solve=(Func<IReadOnlyList<Vector2>,float,Vector2[]>)Delegate.CreateDelegate(typeof(Func<IReadOnlyList<Vector2>,float,Vector2[]>),workspace,type.GetMethod("Solve"));
    start=System.Diagnostics.Stopwatch.GetTimestamp();actual=solve(goals,radius);result.secondFirstRunMs=Ms(start);result.secondUsedBurst=(bool)type.GetProperty("LastRunUsedBurst").GetValue(workspace);Exact(expected,actual);Assert.IsTrue(result.secondUsedBurst);result.exactBits=true;
   }finally{if(workspace!=null)((IDisposable)workspace).Dispose();}
   result.disposed=true;var flags=System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;
   result.created=(int)type.GetProperty("CreatedWorkspaces",flags).GetValue(null);result.disposedCount=(int)type.GetProperty("DisposedWorkspaces",flags).GetValue(null);result.active=(int)type.GetProperty("ActiveWorkspaces",flags).GetValue(null);result.burstCalls=(int)type.GetProperty("BurstCalls",flags).GetValue(null);result.managedCalls=(int)type.GetProperty("ManagedNativeCalls",flags).GetValue(null);
   Assert.AreEqual(2,result.created);Assert.AreEqual(2,result.disposedCount);Assert.AreEqual(0,result.active);Assert.AreEqual(14,result.burstCalls);Assert.AreEqual(0,result.managedCalls);
   views.Add(Cache("test-finished"));result.cacheViews=views.ToArray();result.utcEnd=DateTime.UtcNow.ToString("o");System.IO.File.WriteAllText(System.IO.Path.Combine(Output(),"burst-startup.json"),JsonUtility.ToJson(result,true));
  }
 }
}
