  static int Bin(int n) { int i=0;while(n>1&&i<9){n>>=1;i++;}return i; }
  static void Exact(Vector2[] a,Vector2[] b) { typeof(P3UniformOutputTests).GetMethod("Exact",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Static).Invoke(null,new object[]{a,b}); }
  static bool Route(TerrainNavigationGrid nav, uint[] walkable, byte[] edges,int from,int to,Counts c,int[] edgeHits) {
   if(from<0||to<0||from>=nav.CellCount||to>=nav.CellCount||walkable[from]==0||walkable[to]==0)return false;
   if(from==to)return true;
   int step,bit;
   if(from/nav.ResolutionX==to/nav.ResolutionX){step=to>from?1:-1;bit=step>0?0:4;}
   else if(from%nav.ResolutionX==to%nav.ResolutionX){step=to>from?nav.ResolutionX:-nav.ResolutionX;bit=step>0?1:5;}
   else return false;
   int requested=Math.Abs((to-from)/step);c.requestedRouteEdges+=requested;c.routeLengthBins[Bin(requested)]++;
   int tested=0;bool clear=true;
   for(int cell=from;cell!=to;cell+=step){c.routeEdges++;tested++;edgeHits[cell*8+bit]++;if((edges[cell]&(1<<bit))==0){clear=false;break;}}
   c.maxRouteEdges=Math.Max(c.maxRouteEdges,tested);return clear;
  }
  static Counts Count(TerrainNavigationGrid nav,Vector2[] goals,Vector2[] directions,float stopRadius) {
   var c=new Counts{cells=directions.Length,goals=goals.Length};
   var rows=new List<int>[nav.ResolutionZ];var columns=new List<int>[nav.ResolutionX];Vector2 mean=Vector2.zero;int total=0;
   var unique=new HashSet<int>();
   foreach(var p in goals){if(!nav.TryGetCell(p,out int cell)||!nav.IsWalkable(p))continue;int x=cell%nav.ResolutionX,z=cell/nav.ResolutionX;if(rows[z]==null)rows[z]=new List<int>();if(columns[x]==null)columns[x]=new List<int>();rows[z].Add(cell);columns[x].Add(cell);mean+=nav.CellCenter(cell);total++;unique.Add(cell);}
   c.validGoals=total;c.uniqueGoalCells=unique.Count;
   if(total==0)return c;mean/=total;
   foreach(var list in rows)if(list!=null){list.Sort();c.rowLists++;c.maxRowGoals=Math.Max(c.maxRowGoals,list.Count);}
   foreach(var list in columns)if(list!=null){list.Sort();c.columnLists++;c.maxColumnGoals=Math.Max(c.maxColumnGoals,list.Count);}
   var flags=System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance;
   var walkable=(uint[])typeof(TerrainNavigationGrid).GetField("walkable",flags).GetValue(nav);
   var edges=(byte[])typeof(TerrainNavigationGrid).GetField("neighbours",flags).GetValue(nav);
   var hits=new int[nav.CellCount*8];
   float near=Mathf.Max(nav.CellSize*4,stopRadius+nav.CellSize);
   for(int cell=0;cell<directions.Length;cell++){
    Vector2 original=directions[cell];if(original.sqrMagnitude<.0001f){c.inactive++;continue;}c.active++;
    if(rows[cell/nav.ResolutionX]==null&&columns[cell%nav.ResolutionX]==null)c.bothLanesEmpty++;
    Vector2 here=nav.CellCenter(cell),toward=mean-here;bool horizontal=Mathf.Abs(toward.x)>=Mathf.Abs(toward.y);int sign=(horizontal?toward.x:toward.y)>=0?1:-1;
    if(horizontal)c.horizontal++;else c.vertical++;
    Vector2 proposed=horizontal?new Vector2(sign,0):new Vector2(0,sign);if(Vector2.Dot(proposed,original)<.5f){c.dotRejected++;continue;}
    var list=horizontal?rows[cell/nav.ResolutionX]:columns[cell%nav.ResolutionX];if(list==null){c.noList++;continue;}
    c.searches++;c.searchSizeBins[Bin(list.Count)]++;
    int lo=0,hi=list.Count-1,probe=0;bool found=false;
    while(lo<=hi){int mid=lo+((hi-lo)>>1);c.searchComparisons++;int v=list[mid];if(v==cell){probe=mid;found=true;break;}if(v<cell)lo=mid+1;else hi=mid-1;}
    if(!found)probe=~lo;
    int at=list.BinarySearch(cell);Assert.AreEqual(at,probe,"Search clone index including duplicate-hit choice");
    if(at<0)at=~at;else{c.targetSelf++;continue;}int pick=sign>0?at:at-1;if(pick<0||pick>=list.Count){c.noForwardTarget++;continue;}int target=list[pick];
    if(Vector2.Distance(here,nav.CellCenter(target))<near){c.tooNear++;continue;}
    c.routeCalls++;bool clear=Route(nav,walkable,edges,cell,target,c,hits);Assert.AreEqual(nav.HasClearCardinalRoute33(cell,target),clear,"Route decision");
    if(!clear){c.routeBlocked++;continue;}c.routeClear++;c.replacements++;
    if(original.x!=proposed.x||original.y!=proposed.y)c.changed++;
    directions[cell]=proposed;
   }
   foreach(int hit in hits)if(hit>0){c.uniqueDirectedEdges++;c.maxEdgeReuse=Math.Max(c.maxEdgeReuse,hit);}
   Assert.AreEqual(c.cells,c.inactive+c.active);
   Assert.AreEqual(c.active,c.dotRejected+c.noList+c.targetSelf+c.noForwardTarget+c.tooNear+c.routeBlocked+c.replacements);
   Assert.AreEqual(c.searches,c.targetSelf+c.noForwardTarget+c.tooNear+c.routeCalls);
   return c;
  }
